// ═══════════════════════════════════════════════════════════════════
//  UART — Master → Slave  (12-byte fixed packet)
// ═══════════════════════════════════════════════════════════════════

uint8_t calcChecksum(uint8_t cmd, float lat, float lng, uint8_t alt) {
  uint8_t cs = cmd;
  uint8_t* lb = (uint8_t*)&lat;
  uint8_t* lo = (uint8_t*)&lng;
  for (int i = 0; i < 4; i++) { cs ^= lb[i]; cs ^= lo[i]; }
  cs ^= alt;
  return cs;
}

void sendUartPacket(uint8_t cmd, float lat, float lng, uint8_t alt) {
  uint8_t pkt[12];
  pkt[0]  = UART_START;
  pkt[1]  = cmd;
  memcpy(&pkt[2], &lat, 4);
  memcpy(&pkt[6], &lng, 4);
  pkt[10] = alt;
  pkt[11] = calcChecksum(cmd, lat, lng, alt);
  Serial2.write(pkt, 12);
  Serial.printf("[UART TX] CMD=0x%02X lat=%.6f lng=%.6f alt=%d cs=0x%02X\n",
                cmd, lat, lng, alt, pkt[11]);
}

// ═══════════════════════════════════════════════════════════════════
//  MQTT status publishing
//
//  includePolygon = true  → sent ONCE at mission start.
//                           Embeds the full flight_path array so the
//                           cockpit Mapbox map can draw the route.
//  includePolygon = false → lightweight update on every waypoint tick.
//                           Only state + current_index + total_points.
// ═══════════════════════════════════════════════════════════════════

const char* statusName(MissionStatus s) {
  switch (s) {
    case IDLE:    return "IDLE";
    case FLYING:  return "FLYING";
    case RTH:     return "RTH";
    case LANDING: return "LANDING";
    case ABORTED: return "ABORTED";
    default:      return "IDLE";
  }
}

void publishMissionStatus(bool includePolygon = false) {
  if (!mqttConnected) return;

  // FIX #3: warn when caller asked for full polygon but mission has 0 points
  if (includePolygon && buf3.total_points == 0) {
    Serial.println(F("[MQTT] WARNING: includePolygon=true but total_points==0 — sending lightweight status"));
  }

  if (!includePolygon || buf3.total_points == 0) {
    // ── Lightweight update ───────────────────────────────────────
    char payload[128];
    snprintf(payload, sizeof(payload),
      "{\"state\":\"%s\",\"current_index\":%d,\"total_points\":%d,\"mission_type\":%d}",
      statusName(buf3.status),
      buf3.current_index,
      buf3.total_points,
      buf3.mission_type);
    mqttClient.publish(topic(SUFFIX_STATUS).c_str(), payload);
    Serial.printf("[MQTT] Status: %s\n", payload);
    return;
  }

  // ── Full polygon payload ─────────────────────────────────────────
  // Builds: {"state":"FLYING","current_index":0,"total_points":N,
  //          "mission_type":1,"polygon":[{"lat":x,"lng":y},…]}
  //
  // Max size estimate: base ~80 chars + N * ~30 chars per point.
  // For 100 waypoints ≈ 3 080 bytes — well within MQTT_BUFFER_SIZE.
  //
  // We use a dynamic String here rather than a fixed char[] to avoid
  // stack overflow for large missions.

  String payload = "{\"state\":\"";
  payload += statusName(buf3.status);
  payload += "\",\"current_index\":0,\"total_points\":";
  payload += buf3.total_points;
  payload += ",\"mission_type\":";
  payload += buf3.mission_type;
  payload += ",\"polygon\":[";

  for (uint16_t i = 0; i < buf3.total_points; i++) {
    if (i > 0) payload += ",";
    payload += "{\"lat\":";
    payload += String(missionWaypoints[i].lat, 6);
    payload += ",\"lng\":";
    payload += String(missionWaypoints[i].lng, 6);
    payload += "}";
  }
  payload += "]}";

  bool ok = mqttClient.publish(topic(SUFFIX_STATUS).c_str(), payload.c_str(), payload.length());
  Serial.printf("[MQTT] Status+polygon: %d pts, %d bytes, ok=%d\n",
                buf3.total_points, payload.length(), ok);
}

// Lightweight state-only publish (ARM/DISARM etc.)
void publishState(const char* state) {
  if (!mqttConnected) return;
  char payload[48];
  snprintf(payload, sizeof(payload), "{\"state\":\"%s\"}", state);
  mqttClient.publish(topic(SUFFIX_STATUS).c_str(), payload);
}

void publishAck(bool success, const String& message) {
  char buf[160];
  snprintf(buf, sizeof(buf), "{\"success\":%s,\"message\":\"%s\"}",
           success ? "true" : "false", message.c_str());
  mqttClient.publish(topic(SUFFIX_SD_ACK).c_str(), buf);
}

// ═══════════════════════════════════════════════════════════════════
//  Waypoint serving — publishes lightweight status on every point
// ═══════════════════════════════════════════════════════════════════

void serveWaypoint(uint16_t index) {
  if (index >= buf3.total_points) return;
  Waypoint& wp = missionWaypoints[index];

  buf1.lat = wp.lat;
  buf1.lng = wp.lng;
  buf1.alt = wp.alt;

  buf3.current_index = index;

  sendUartPacket(CMD_WAYPOINT, wp.lat, wp.lng, wp.alt);

  // Publish progress update so cockpit map advances the progress line
  publishMissionStatus(false);

  Serial.printf("[MISSION] Serving waypoint %d/%d (%.6f, %.6f, %dm)\n",
                index + 1, buf3.total_points, wp.lat, wp.lng, wp.alt);
}

// ═══════════════════════════════════════════════════════════════════
//  UART RX — Slave → Master
// ═══════════════════════════════════════════════════════════════════

uint8_t  rxBuf[32];
uint8_t  rxLen    = 0;
bool     rxActive = false;

void handleSlaveUart() {
  while (Serial2.available()) {
    uint8_t b = Serial2.read();

    if (!rxActive) {
      switch (b) {

        case RESP_NEXT_REQ: {
          Serial.println(F("[UART RX] NEXT_REQUEST received"));

          if (rthRequested) {
            rthRequested = false;
            if (buf2.set) {
              Serial.println(F("[MISSION] RTH: serving home point"));
              buf3.status = RTH;
              publishMissionStatus(false);
              sendUartPacket(CMD_WAYPOINT, buf2.lat, buf2.lng, buf2.alt);
            } else {
              Serial.println(F("[MISSION] RTH requested but no home point — ABORT"));
              triggerAbort();
            }
            break;
          }

          if (abortRequested) {
            abortRequested = false;
            triggerAbort();
            break;
          }

          if (buf3.status == FLYING) {
            uint16_t next = buf3.current_index + 1;
            if (next < buf3.total_points) {
              serveWaypoint(next);
            } else {
              Serial.println(F("[MISSION] Last waypoint reached"));
              if (queueActive) {
                advanceQueue();
              } else {
                buf3.status = LANDING;
                publishMissionStatus(false);
                Serial.println(F("[MISSION] Mission complete"));
              }
            }
          }
          break;
        }

        case RESP_HOME_POINT: {
          if (!buf2.set) {
            rxBuf[0]  = b;
            rxLen     = 1;
            rxActive  = true;
            Serial.println(F("[UART RX] HOME_POINT incoming, collecting 9 bytes..."));
          }
          break;
        }

        case RESP_ARRIVED_HOME: {
          Serial.println(F("[UART RX] ARRIVED_HOME — drone is at home position"));
          buf3.status = LANDING;
          publishMissionStatus(false);
          sendUartPacket(CMD_LAND, buf2.lat, buf2.lng, 0);
          break;
        }

        case RESP_LANDED: {
          Serial.println(F("[UART RX] LANDED — drone on the ground"));
          buf3.status = IDLE;
          // FIX #4: clear home point so a subsequent slave reboot will
          // always push a fresh GPS-derived home before next mission.
          buf2.set    = false;
          clearMissionRam();
          publishMissionStatus(false);
          publishState("IDLE");
          break;
        }

        case RESP_NACK: {
          Serial.println(F("[UART RX] NACK from slave — last command rejected"));
          break;
        }

        case RESP_TELEMETRY: {
          rxBuf[0] = b;
          rxLen    = 1;
          rxActive = true;
          break;
        }

        default:
          break;
      }
    }

    else {
      rxBuf[rxLen++] = b;

      // HOME_POINT: 0x07(1) + lat(4) + lng(4) + alt(1) = 10 bytes
      if (rxBuf[0] == RESP_HOME_POINT && rxLen == 10) {
        rxActive = false;
        float   homeLat, homeLng;
        uint8_t homeAlt;
        memcpy(&homeLat, &rxBuf[1], 4);
        memcpy(&homeLng, &rxBuf[5], 4);
        homeAlt = rxBuf[9];
        buf2.lat = homeLat;
        buf2.lng = homeLng;
        buf2.alt = homeAlt;
        buf2.set = true;
        Serial.printf("[BUFFER2] Home set: %.6f, %.6f, %dm\n", homeLat, homeLng, homeAlt);
        rxLen = 0;
      }

      // TELEMETRY: 0x0A(1) + 23 payload bytes = 24 bytes total
      if (rxBuf[0] == RESP_TELEMETRY && rxLen == 24) {
        rxActive = false;
        uint8_t* p = &rxBuf[1];
        memcpy(&latestTelemetry.lat,             p, 4); p += 4;
        memcpy(&latestTelemetry.lng,             p, 4); p += 4;
        memcpy(&latestTelemetry.altitude,        p, 4); p += 4;
        memcpy(&latestTelemetry.speed,           p, 4); p += 4;
        memcpy(&latestTelemetry.heading,         p, 4); p += 4;
        latestTelemetry.battery         = *p++;
        memcpy(&latestTelemetry.signal_strength, p, 2);
        telemetryReady = true;
        Serial.printf("[UART RX] Telemetry: lat=%.6f lng=%.6f alt=%.1f bat=%d%%\n",
          latestTelemetry.lat, latestTelemetry.lng,
          latestTelemetry.altitude, latestTelemetry.battery);
        rxLen = 0;
      }
    }
  }
}

// ═══════════════════════════════════════════════════════════════════
//  Mission RAM helpers

void clearMissionRam() {
  buf3.current_index = 0;
  buf3.total_points  = 0;
  buf3.mission_type  = 0;
  memset(missionWaypoints, 0, sizeof(missionWaypoints));
}

bool loadMissionIntoRam(const String& missionId) {
  String path = "/missions/m_" + missionId + ".json";
  if (!SD.exists(path.c_str())) {
    Serial.println("[MISSION] File not found: " + path);
    return false;
  }

  File file = SD.open(path.c_str());
  if (!file) {
    Serial.println(F("[MISSION] Failed to open mission file"));
    return false;
  }

  JsonDocument doc;
  DeserializationError err = deserializeJson(doc, file);
  file.close();

  if (err) {
    Serial.println(F("[MISSION] JSON parse error"));
    return false;
  }

  buf3.mission_type = doc["mission_type"] | 0;
  String name       = doc["name"] | "?";

  JsonArray path_arr = doc["flight_path"].as<JsonArray>();
  uint16_t count = 0;
  for (JsonObject wp : path_arr) {
    if (count >= MAX_WAYPOINTS) break;
    missionWaypoints[count].lat   = wp["lat"]   | 0.0f;
    missionWaypoints[count].lng   = wp["lng"]   | 0.0f;
    missionWaypoints[count].alt   = (uint8_t)(wp["alt"]   | 0);
    missionWaypoints[count].speed = (uint8_t)(wp["speed"] | 0);
    count++;
  }

  buf3.total_points  = count;
  buf3.current_index = 0;

  Serial.printf("[MISSION] Loaded '%s' — %d waypoints, type=%d\n",
                name.c_str(), count, buf3.mission_type);
  return true;
}

// ═══════════════════════════════════════════════════════════════════
//  Mission / Queue execution
// ═══════════════════════════════════════════════════════════════════

// ── KEY CHANGE: startMissionFlight publishes with polygon=true ────
// This is the ONE moment where the full route is sent to the app.
// The cockpit map stores it and redraws it. Every subsequent update
// only needs current_index (no polygon resent to save bandwidth).
void startMissionFlight() {
  buf3.status        = FLYING;
  buf3.current_index = 0;
  rthRequested       = false;
  abortRequested     = false;

  // Send full polygon so cockpit map can draw the entire route
  publishMissionStatus(true);

  publishState("FLYING");

  serveWaypoint(0);
}

void executeMissionFromSd(const String& missionId) {
  if (!sdReady) { publishAck(false, "SD not ready"); return; }

  clearMissionRam();
  bool ok = loadMissionIntoRam(missionId);

  if (!ok) { publishAck(false, "Mission " + missionId + " not found"); return; }

  SD.end();
  sdReady = false;
  Serial.println(F("[SD] Unmounted for flight"));

  queueActive = false;
  publishAck(true, "Mission executing");
  startMissionFlight();
}

void executeQueueFromSd(const String& filename) {
  if (!sdReady) { publishAck(false, "SD not ready"); return; }

  String path = "/queues/" + filename;
  if (!SD.exists(path.c_str())) { publishAck(false, "Queue not found"); return; }

  File file = SD.open(path.c_str());
  JsonDocument doc;
  deserializeJson(doc, file);
  file.close();

  JsonArray missions = doc["missions"].as<JsonArray>();
  queueTotal = 0;
  for (JsonObject m : missions) {
    if (queueTotal >= MAX_QUEUE_MISSIONS) break;
    queueMissionIds[queueTotal++] = m["id"].as<String>();
  }

  if (queueTotal == 0) { publishAck(false, "Queue is empty"); return; }

  queueCurrentIdx = 0;
  queueActive     = true;

  Serial.printf("[QUEUE] Starting queue '%s' — %d missions\n",
                doc["name"] | "?", queueTotal);

  clearMissionRam();
  bool ok = loadMissionIntoRam(queueMissionIds[0]);
  if (!ok) { publishAck(false, "First mission not found on SD"); queueActive = false; return; }

  SD.end();
  sdReady = false;
  Serial.println(F("[SD] Unmounted for queue flight"));

  publishAck(true, "Queue executing");
  startMissionFlight();   // sends polygon for first mission
}

void advanceQueue() {
  // FIX #5: was recursive — replaced with iterative loop to avoid
  // stack overflow when multiple consecutive missions are missing.
  while (true) {
    queueCurrentIdx++;

    if (queueCurrentIdx >= queueTotal) {
      Serial.println(F("[QUEUE] All missions complete"));
      queueActive = false;
      buf3.status = LANDING;
      publishMissionStatus(false);
      publishState("IDLE");
      return;
    }

    Serial.printf("[QUEUE] Advancing to mission %d/%d\n",
                  queueCurrentIdx + 1, queueTotal);

    sdReady = SD.begin(SD_CS_PIN);
    if (!sdReady) {
      Serial.println(F("[SD] Remount failed during queue advance — aborting queue"));
      publishAck(false, "SD remount failed during queue advance");
      queueActive = false;
      return;
    }

    clearMissionRam();
    bool ok = loadMissionIntoRam(queueMissionIds[queueCurrentIdx]);

    SD.end();
    sdReady = false;

    if (!ok) {
      Serial.println("[QUEUE] Mission not found: " + queueMissionIds[queueCurrentIdx]);
      // iterate — try next mission in queue
      continue;
    }

    // startMissionFlight sends new polygon for the next queue mission
    startMissionFlight();
    return;
  }
}

void triggerAbort() {
  Serial.println(F("[MISSION] ABORT — landing in place"));
  buf3.status = ABORTED;
  queueActive = false;
  publishMissionStatus(false);
  publishState("LANDING");
  sendUartPacket(CMD_LAND, buf1.lat, buf1.lng, 0);
}

void handleStopCommand() {
  Serial.println(F("[MISSION] STOP received"));
  abortRequested = false;
  rthRequested   = false;
  queueActive    = false;
  buf3.status    = IDLE;
  clearMissionRam();
  publishMissionStatus(false);
  publishState("IDLE");
  sendUartPacket(CMD_ABORT, 0, 0, 0);
}

// ═══════════════════════════════════════════════════════════════════
//  Telemetry (Buffer 4 → MQTT + HTTP)
//
//  CHANGE: pushTelemetryMQTT now appends current_index + total_points
//  so the cockpit map progress line updates on every telemetry tick
//  (every ~3 s) in addition to the waypoint-point events.
// ═══════════════════════════════════════════════════════════════════

void pushTelemetryMQTT() {
  if (!mqttConnected) return;
  char payload[220];
  snprintf(payload, sizeof(payload),

// ═══════════════════════════════════════════════════════════════════
//  SD File Operations

void ensureDir(const char* path) {
  if (!SD.exists(path)) SD.mkdir(path);
}

bool remountSd() {
  if (sdReady) return true;
  sdReady = SD.begin(SD_CS_PIN);
  if (sdReady) {
    ensureDir("/missions");
    ensureDir("/queues");
  }
  return sdReady;
}

void saveMissionToSd(const String& jsonPayload) {
  if (!remountSd()) { publishAck(false, "SD not available"); return; }
  JsonDocument doc;
  deserializeJson(doc, jsonPayload);
  String id   = doc["id"].as<String>();
  String name = doc["name"].as<String>();
  String path = "/missions/m_" + id + ".json";
  if (SD.exists(path.c_str())) SD.remove(path.c_str());
  File file = SD.open(path.c_str(), FILE_WRITE);
  if (!file) { publishAck(false, "Failed to open file for mission " + name); return; }
  int total = jsonPayload.length(), offset = 0;
  while (offset < total) {
    int chunk = min(512, total - offset);
    file.write((uint8_t*)(jsonPayload.c_str() + offset), chunk);
    offset += chunk;
  }
  file.close();
  Serial.println("[SD] Saved: " + path);
  publishAck(true, "Mission " + name + " saved to SD");
}

void deleteMissionFromSd(const String& id) {
  if (!remountSd()) { publishAck(false, "SD not available"); return; }
  String path = "/missions/m_" + id + ".json";
  if (SD.exists(path.c_str())) {
    SD.remove(path.c_str());
    publishAck(true, "Mission " + id + " deleted");
  } else {
    publishAck(false, "Mission file not found");
  }
}

void saveQueueToSd(const String& jsonPayload) {
  if (!remountSd()) { publishAck(false, "SD not available"); return; }
  JsonDocument doc;
  deserializeJson(doc, jsonPayload);
  String name     = doc["name"].as<String>();
  String filename = doc["filename"].as<String>();
  String path = "/queues/" + filename;
  if (SD.exists(path.c_str())) SD.remove(path.c_str());
  File file = SD.open(path.c_str(), FILE_WRITE);
  if (file) {
    file.print(jsonPayload);
    file.close();
    publishAck(true, "Queue " + name + " saved");
  } else {
    publishAck(false, "Failed to save queue " + name);
  }
}

void deleteQueueFromSd(const String& filename) {
  if (!remountSd()) { publishAck(false, "SD not available"); return; }
  String path = "/queues/" + filename;
  if (SD.exists(path.c_str())) {
    SD.remove(path.c_str());
    publishAck(true, "Queue deleted");
  } else {
    publishAck(false, "Queue file not found");
  }
}

void sendSdList() {
  if (!remountSd()) {
    mqttClient.publish(topic(SUFFIX_SD_LIST_RESP).c_str(), "{\"missions\":[],\"queues\":[]}");
    return;
  }

  String missionsJson = "", queuesJson = "";

  JsonDocument filter;
  filter["id"] = true; filter["name"] = true;
  filter["waypoint_count"] = true; filter["estimated_time"] = true;
  filter["total_distance"] = true; filter["filename"] = true;
  filter["missions"] = true;

  File dir = SD.open("/missions");
  if (dir) {
    File entry = dir.openNextFile();
    while (entry) {
      if (!entry.isDirectory()) {
        JsonDocument doc;
        if (deserializeJson(doc, entry, DeserializationOption::Filter(filter)) == DeserializationError::Ok) {
          if (missionsJson.length() > 0) missionsJson += ",";
          missionsJson += "{\"id\":\""          + doc["id"].as<String>()             + "\","
                        + "\"name\":\""         + doc["name"].as<String>()           + "\","
                        + "\"waypoint_count\":" + doc["waypoint_count"].as<String>() + ","
                        + "\"estimated_time\":" + doc["estimated_time"].as<String>() + ","
                        + "\"total_distance\":" + doc["total_distance"].as<String>() + "}";
        }
      }
      entry.close();
      entry = dir.openNextFile();
    }
    dir.close();
  }

  dir = SD.open("/queues");
  if (dir) {
    File entry = dir.openNextFile();
    while (entry) {
      if (!entry.isDirectory()) {
        JsonDocument doc;
        deserializeJson(doc, entry, DeserializationOption::Filter(filter));
        if (queuesJson.length() > 0) queuesJson += ",";
        // FIX #7: doc["missions"] is a JSON array — must use serializeJson,
        // not .as<String>() which returns "null" for array types in ArduinoJson.
        String missionsArray = "[]";
        if (doc["missions"].is<JsonArray>()) {
          serializeJson(doc["missions"], missionsArray);
        }
        queuesJson += "{\"name\":\""    + doc["name"].as<String>()    + "\","
                    + "\"filename\":\"" + doc["filename"].as<String>() + "\","
                    + "\"missions\":"   + missionsArray + "}";
      }
      entry.close();
      entry = dir.openNextFile();
    }
    dir.close();
  }

  String response = "{\"missions\":[" + missionsJson + "],\"queues\":[" + queuesJson + "]}";
  mqttClient.publish(topic(SUFFIX_SD_LIST_RESP).c_str(), response.c_str());
  Serial.println(F("[SD] List sent"));
}

// ═══════════════════════════════════════════════════════════════════
//  HTTP helpers
// ═══════════════════════════════════════════════════════════════════

String postJSON(const char* url, String body) {
  HTTPClient http;
  http.begin(url);
  http.addHeader(F("Content-Type"), F("application/json"));
  int code = http.POST(body);
  String response = "";
  if (code == 200) response = http.getString();
  http.end();
  return response;
}

void fetchSlot(int index) {
  char body[80];
  snprintf(body, sizeof(body),
    "{\"drone_id\":%s,\"user_id\":%s,\"slot_index\":%d}",
    droneID.c_str(), userID.c_str(), index);
  String response = postJSON(SLOT_URL, String(body));

  JsonDocument doc;
  DeserializationError err = deserializeJson(doc, response);

  if (err || doc["success"] != true) {
    Serial.println("[SLOT] Error or parse fail: " + response);
    return;
  }

  // FIX #8: slot response was parsed but never acted on.
  // Now we check whether the slot carries a mission or a queue
  // and dispatch accordingly — same path as an MQTT execute command.
  Serial.printf("[SLOT] Received slot %d\n", index);

  String type       = doc["type"] | "";
  String missionId  = doc["mission_id"] | "";
  String filename   = doc["filename"]   | "";

  if (type == "mission" && missionId.length() > 0) {
    Serial.println("[SLOT] Executing mission from slot: " + missionId);
    executeMissionFromSd(missionId);
  } else if (type == "queue" && filename.length() > 0) {
    Serial.println("[SLOT] Executing queue from slot: " + filename);
    executeQueueFromSd(filename);
  } else {
    Serial.println(F("[SLOT] No executable payload in slot response"));
  }
}

// ═══════════════════════════════════════════════════════════════════
//  MQTT Callback
// ═══════════════════════════════════════════════════════════════════

void mqttCallback(char* topic_c, byte* payload, unsigned int length) {
  String topicStr = String(topic_c);
  String msg((char*)payload, length);

  Serial.print(F("[MQTT IN] ")); Serial.print(topicStr);
  Serial.print(F(" : ")); Serial.println(msg);

  JsonDocument doc;
  deserializeJson(doc, msg);

  if (topicStr == topic(SUFFIX_SD_LIST))         { sendSdList();                                    return; }
  if (topicStr == topic(SUFFIX_SD_SAVE_MISSION)) { saveMissionToSd(msg);                            return; }
  if (topicStr == topic(SUFFIX_SD_DEL_MISSION))  { deleteMissionFromSd(doc["id"].as<String>());     return; }
  if (topicStr == topic(SUFFIX_SD_SAVE_QUEUE))   { saveQueueToSd(msg);                              return; }
  if (topicStr == topic(SUFFIX_SD_DEL_QUEUE))    { deleteQueueFromSd(doc["filename"].as<String>()); return; }

  if (topicStr == topic(SUFFIX_SD_EXEC_MISSION)) {
    String command = doc["command"].as<String>();
    if (command == "STOP") {
      handleStopCommand();
    } else {
      String id = doc["id"].as<String>();
      if (id.length() > 0) executeMissionFromSd(id);
    }
    return;
  }

  if (topicStr == topic(SUFFIX_SD_EXEC_QUEUE)) {
    executeQueueFromSd(doc["filename"].as<String>());
    return;
  }

  if (topicStr == topic(SUFFIX_COMMAND)) {
    String cmd = doc["command"].as<String>();

    if (cmd == "ARM") {
      publishState("ARMED");
    } else if (cmd == "DISARM") {
      publishState("IDLE");
    } else if (cmd == "TAKEOFF") {
      publishState("FLYING");
    } else if (cmd == "LAND") {
      Serial.println(F("[CMD] LAND — descending at current position"));
      buf3.status = LANDING;
      publishMissionStatus(false);
      sendUartPacket(CMD_LAND, buf1.lat, buf1.lng, 0);
    } else if (cmd == "RTH") {
      Serial.println(F("[CMD] RTH requested"));
      rthRequested = true;
      buf3.status  = RTH;
      publishMissionStatus(false);
      if (buf3.total_points == 0 && buf2.set) {
        sendUartPacket(CMD_WAYPOINT, buf2.lat, buf2.lng, buf2.alt);
      }
    } else if (cmd == "ABORT") {
      Serial.println(F("[CMD] ABORT"));
      triggerAbort();
    } else if (cmd == "LOITER") {
      Serial.println(F("[CMD] LOITER — pausing point serving"));
      buf3.status = IDLE;
      publishMissionStatus(false);
      publishState("FLYING");
    }
    return;
  }

  if (topicStr == topic(SUFFIX_MISSION)) {
    String missionId = doc["mission_id"].as<String>();
    Serial.println("[MQTT] Direct mission push, ID: " + missionId);
    char ack[100];
    snprintf(ack, sizeof(ack),
      "{\"success\":true,\"message\":\"Mission received\",\"mission_id\":\"%s\"}",
      missionId.c_str());
    mqttClient.publish(topic(SUFFIX_ACK).c_str(), ack);
    return;
  }
}

// ═══════════════════════════════════════════════════════════════════
//  MQTT Connect
// ═══════════════════════════════════════════════════════════════════

void connectMQTT() {
  secureClient.setInsecure();
  mqttClient.setServer(MQTT_HOST, MQTT_PORT);
  mqttClient.setCallback(mqttCallback);
  mqttClient.setBufferSize(MQTT_BUFFER_SIZE);

  String clientId = "esp32_" + droneID;
  if (mqttClient.connect(clientId.c_str(), MQTT_USER, MQTT_PASS)) {
    mqttConnected = true;
    Serial.println(F("[MQTT] Connected!"));
    mqttClient.subscribe(topic(SUFFIX_COMMAND).c_str());
    mqttClient.subscribe(topic(SUFFIX_MISSION).c_str());
    mqttClient.subscribe(topic(SUFFIX_SD_LIST).c_str());
    mqttClient.subscribe(topic(SUFFIX_SD_SAVE_MISSION).c_str());
    mqttClient.subscribe(topic(SUFFIX_SD_DEL_MISSION).c_str());
    mqttClient.subscribe(topic(SUFFIX_SD_SAVE_QUEUE).c_str());
    mqttClient.subscribe(topic(SUFFIX_SD_DEL_QUEUE).c_str());
    mqttClient.subscribe(topic(SUFFIX_SD_EXEC_MISSION).c_str());
    mqttClient.subscribe(topic(SUFFIX_SD_EXEC_QUEUE).c_str());
    publishMissionStatus(false);
  } else {
    Serial.print(F("[MQTT] Failed, rc="));
    Serial.println(mqttClient.state());
  }
}

// ═══════════════════════════════════════════════════════════════════
//  BLE — WiFi provisioning
// ═══════════════════════════════════════════════════════════════════

class MyServerCallbacks : public BLEServerCallbacks {
  void onConnect(BLEServer* server) override {
    bleClientConnected = true;
    Serial.println(F("[BLE] App connected"));
  }
  void onDisconnect(BLEServer* server) override {
    bleClientConnected = false;
    Serial.println(F("[BLE] App disconnected"));
    BLEDevice::startAdvertising();
  }
};

class MyCharacteristicCallbacks : public BLECharacteristicCallbacks {
  void onWrite(BLECharacteristic* characteristic) override {
    String value = characteristic->getValue().c_str();
    Serial.print(F("[BLE] Received: ")); Serial.println(value);

    JsonDocument doc;
    DeserializationError err = deserializeJson(doc, value);
    if (err) {
      characteristic->setValue("ERROR:invalid_json");
      characteristic->notify();
      return;
    }

    String command = doc["command"].as<String>();
    if (command == "scan_wifi") {
      WiFi.mode(WIFI_STA);
      WiFi.disconnect();
      delay(100);
      int n = WiFi.scanNetworks();

      String json = "WIFI_LIST:[";
      int added = 0;
      for (int i = 0; i < n; i++) {
        String ssidVal = WiFi.SSID(i);
        if (ssidVal.length() == 0) continue;
        bool isOpen = (WiFi.encryptionType(i) == WIFI_AUTH_OPEN);
        ssidVal.replace("\"", "\\\"");
        if (added > 0) json += ",";
        json += "{\"ssid\":\"" + ssidVal + "\","
              + "\"rssi\":"    + String(WiFi.RSSI(i)) + ","
              + "\"open\":"    + (isOpen ? "true" : "false") + "}";
        added++;
      }
      json += "]";
      WiFi.scanDelete();

      int offset = 0, total = json.length();
      while (offset < total) {
        int end = min(offset + 180, total);
        characteristic->setValue(json.substring(offset, end).c_str());
        characteristic->notify();
        delay(50);
        offset = end;
      }
      Serial.printf("[BLE] WiFi list sent (%d networks)\n", added);
      return;
    }

    receivedSSID     = doc["ssid"].as<String>();
    receivedPassword = doc["password"].as<String>();

    if (receivedSSID.length() == 0) {
      characteristic->setValue("ERROR:missing_ssid");
      characteristic->notify();
      return;
    }

    characteristic->setValue("INFO:connecting");
    characteristic->notify();

    WiFi.mode(WIFI_STA);
    WiFi.begin(receivedSSID.c_str(), receivedPassword.c_str());

    int tries = 0;
    while (WiFi.status() != WL_CONNECTED && tries < 24) { delay(500); tries++; }

    if (WiFi.status() == WL_CONNECTED) {
      preferences.begin("wifi", false);
      preferences.putString("ssid",     receivedSSID);
      preferences.putString("password", receivedPassword);
      preferences.end();
      characteristic->setValue("OK:wifi_connected");
      characteristic->notify();
      delay(800);
      wifiCredentialsReceived = true;
    } else {
      WiFi.disconnect();
      preferences.begin("wifi", false);
      preferences.clear();
      preferences.end();
      characteristic->setValue("ERROR:wifi_failed");
      characteristic->notify();
    }
  }
};

void startBLESetup() {
  Serial.println(F("[BLE] Starting setup mode..."));

  String mac = WiFi.macAddress();
  mac.replace(":", "");
  String bleName = "ENISOLAR-" + mac.substring(6);

  BLEDevice::init(bleName.c_str());
  bleServer = BLEDevice::createServer();
  bleServer->setCallbacks(new MyServerCallbacks());

  BLEService* service = bleServer->createService(BLE_SERVICE_UUID);
  bleCharacteristic = service->createCharacteristic(
    BLE_CHARACTERISTIC_UUID,
    BLECharacteristic::PROPERTY_READ  |
    BLECharacteristic::PROPERTY_WRITE |
    BLECharacteristic::PROPERTY_NOTIFY
  );
  bleCharacteristic->addDescriptor(new BLE2902());
  bleCharacteristic->setCallbacks(new MyCharacteristicCallbacks());
  bleCharacteristic->setValue("READY");

  service->start();
  BLEAdvertising* adv = BLEDevice::getAdvertising();
  adv->addServiceUUID(BLE_SERVICE_UUID);
  adv->setScanResponse(true);
  BLEDevice::startAdvertising();

  Serial.println("[BLE] Advertising as: " + bleName);

  // FIX #9: was an infinite block — now times out after 5 minutes
  // so a device with no phone nearby reboots cleanly instead of hanging.
  const unsigned long BLE_PROVISION_TIMEOUT_MS = 5UL * 60UL * 1000UL; // 5 min
  unsigned long bleStart = millis();
  while (!wifiCredentialsReceived) {
    delay(100);
    if (millis() - bleStart >= BLE_PROVISION_TIMEOUT_MS) {
      Serial.println(F("[BLE] Provisioning timeout (5 min) — restarting..."));
      BLEDevice::stopAdvertising();
      BLEDevice::deinit(true);
      delay(500);
      ESP.restart();
    }
  }

  WiFi.disconnect(true);
  WiFi.mode(WIFI_OFF);
  delay(200);

  BLEDevice::stopAdvertising();
  BLEDevice::deinit(true);
  delay(500);

  Serial.println(F("[BLE] Setup done. Restarting..."));
  delay(1000);
  ESP.restart();
}

// ═══════════════════════════════════════════════════════════════════
//  SETUP
// ═══════════════════════════════════════════════════════════════════

void setup() {
  Serial.begin(115200);
  delay(1000);
  pinMode(BOOT_BUTTON_PIN, INPUT_PULLUP);

  Serial2.begin(SLAVE_BAUD, SERIAL_8N1, SLAVE_RX_PIN, SLAVE_TX_PIN);
  Serial.println(F("[UART] Serial2 ready for slave FC"));

  preferences.begin("wifi", true);
  String savedSSID     = preferences.getString("ssid", "");
  String savedPassword = preferences.getString("password", "");
  preferences.end();

  Serial.print(F("[WIFI] Saved SSID: '")); Serial.print(savedSSID); Serial.println("'");

  if (savedSSID.length() == 0) {
    Serial.println(F("[BOOT] No credentials. Starting BLE setup..."));
    WiFi.mode(WIFI_OFF);
    delay(100);
    startBLESetup();
    return;
  }

  Serial.print(F("[WIFI] Connecting to: ")); Serial.println(savedSSID);
  WiFi.begin(savedSSID.c_str(), savedPassword.c_str());

  int attempts = 0;
  while (WiFi.status() != WL_CONNECTED && attempts < 20) {
    delay(500); Serial.print(F("."));
    attempts++;
  }

  if (WiFi.status() != WL_CONNECTED) {
    Serial.println(F("\n[WIFI] Failed — clearing credentials, restarting for BLE setup..."));
    preferences.begin("wifi", false);
    preferences.clear();
    preferences.end();
    delay(1000);
    ESP.restart();
    return;
  }

  Serial.print(F("\n[WIFI] Connected! MAC: ")); Serial.println(WiFi.macAddress());
  delay(500);

  Serial.println(F("[SD] Initializing..."));
  SPI.begin();
  delay(100);
  sdReady = false;
  if      (SD.begin())                        sdReady = true;
  else if (SD.begin(SD_CS_PIN))               sdReady = true;
  else if (SD.begin(SD_CS_PIN, SPI, 1000000)) sdReady = true;

  if (sdReady) {
    ensureDir("/missions");
    ensureDir("/queues");
    Serial.println(F("[SD] Ready"));
  } else {
    Serial.println(F("[SD] Failed — continuing without SD"));
  }

  char bootBody[48];
  snprintf(bootBody, sizeof(bootBody), "{\"mac\":\"%s\"}", WiFi.macAddress().c_str());
  String bootResp = postJSON(BOOT_URL, String(bootBody));
  Serial.println("[BOOT] Response: " + bootResp);

  JsonDocument bootDoc;
  if (deserializeJson(bootDoc, bootResp) == DeserializationError::Ok
      && bootDoc["success"] == true) {
    userID   = bootDoc["user_id"].as<String>();
    droneID  = bootDoc["drone_id"].as<String>();
    mqttBase = "drone/" + droneID;
    Serial.printf("[BOOT] userID=%s droneID=%s\n", userID.c_str(), droneID.c_str());
    connectMQTT();
  } else {
    Serial.println(F("[BOOT] Failed — no MQTT"));
  }

  Serial.println(F("[UART] Telemetry will arrive via UART RESP_TELEMETRY (0x0A)"));
}

// ═══════════════════════════════════════════════════════════════════
//  LOOP
// ═══════════════════════════════════════════════════════════════════

void loop() {
  if (userID == "" || droneID == "") return;

  unsigned long now = millis();

  if (mqttConnected) {
    if (!mqttClient.connected()) {
      mqttConnected = false;
      Serial.println(F("[MQTT] Lost — reconnecting..."));
      connectMQTT();
    }
    mqttClient.loop();
  }

  handleSlaveUart();

  if (telemetryReady && (now - lastTelemetryTime >= TELEMETRY_MS)) {
    telemetryReady    = false;
    lastTelemetryTime = now;
    pushTelemetryMQTT();
    pushTelemetryHTTP();
  }

  bool buttonPressed = (digitalRead(BOOT_BUTTON_PIN) == LOW);

  if (buttonPressed && !buttonWasPressed) {
    buttonPressStart = now;
    holdTriggered    = false;
  }

  if (buttonPressed && !holdTriggered && (now - buttonPressStart >= 3000)) {
    holdTriggered = true;
    Serial.println(F("[BTN] 3s hold — clearing WiFi, entering BLE setup..."));
    preferences.begin("wifi", false);
    preferences.clear();
    preferences.end();
    delay(500);
    ESP.restart();
  }

  if (buttonPressed && !holdTriggered && !buttonWasPressed) {
    clickCount++;
    lastClickTime = now;
    windowOpen    = true;
    Serial.print(F("[BTN] Click count: ")); Serial.println(clickCount);
  }
  buttonWasPressed = buttonPressed;

  if (windowOpen && (now - lastClickTime >= CLICK_WINDOW_MS)) {
    windowOpen = false;
    int selected = clickCount;
    clickCount   = 0;
    if (selected >= 1 && selected <= 5) fetchSlot(selected);
  }

  delay(10);
}
