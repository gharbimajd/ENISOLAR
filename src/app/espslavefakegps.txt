// ═══════════════════════════════════════════════════════════════════
//  ENISOLAR — Slave FC Simulator  |  ESP32-WROOM
//
//  PURPOSE: Simulate a real flight controller for testing master logic.
//  This is NOT production FC code. It fakes GPS, IMU, and flight
//  physics so the master can be tested end-to-end without motors.
//
//  UART pins (connect to MASTER crossed):
//    Slave GPIO16 (RX) ← Master GPIO17 (TX)
//    Slave GPIO17 (TX) → Master GPIO16 (RX)
//    GND ──────────────── GND
//
//  UART: receives 12-byte packets from master, sends responses back
//        sends telemetry (0x0A) to master every 3s via same UART
// ═══════════════════════════════════════════════════════════════════

#include <math.h>

// ── UART pins (crossed from master) ──────────────────────────────
#define MASTER_RX_PIN   16
#define MASTER_TX_PIN   17
#define MASTER_BAUD     115200

// ── UART packet bytes (Master → Slave) ───────────────────────────
#define UART_START      0xAA
#define CMD_WAYPOINT    0x01
#define CMD_RTH         0x02
#define CMD_LAND        0x03
#define CMD_ABORT       0x04

// ── UART response bytes (Slave → Master) ─────────────────────────
#define RESP_NEXT_REQ     0x06
#define RESP_HOME_POINT   0x07   // 0x07 + lat(4) + lng(4) + alt(1) = 10 bytes
#define RESP_ARRIVED_HOME 0x08
#define RESP_LANDED       0x09
#define RESP_TELEMETRY    0x0A   // 0x0A + lat(4)+lng(4)+alt(4)+spd(4)+hdg(4)+bat(1)+sig(2) = 24 bytes
#define RESP_NACK         0x15

// ── Timing ────────────────────────────────────────────────────────
#define WAYPOINT_TRAVEL_MS   6000
#define LANDING_DURATION_MS  4000
#define RTH_TRAVEL_MS        8000
#define TELEMETRY_INTERVAL_MS 3000

// ── Simulation tuning ─────────────────────────────────────────────
#define GPS_DRIFT_SCALE      0.000005f
#define BATTERY_DRAIN_PER_WP 1

// ═══════════════════════════════════════════════════════════════════
//  STATE MACHINE
// ═══════════════════════════════════════════════════════════════════

enum SlaveState {
  SIM_IDLE,
  SIM_FLYING,
  SIM_LANDING,
  SIM_RTH,
  SIM_ABORTED
};

SlaveState simState = SIM_IDLE;

// ═══════════════════════════════════════════════════════════════════
//  SIMULATED SENSOR VALUES
// ═══════════════════════════════════════════════════════════════════

float   simLat     = 36.806500f;
float   simLng     = 10.181500f;
float   simAlt     = 0.0f;
float   simSpeed   = 0.0f;
float   simHeading = 0.0f;
uint8_t simBattery = 100;
int16_t simSignal  = -55;

float   targetLat  = 36.806500f;
float   targetLng  = 10.181500f;
float   targetAlt  = 0.0f;

float   homeLat    = 36.806500f;
float   homeLng    = 10.181500f;
uint8_t homeAlt    = 0;

// ═══════════════════════════════════════════════════════════════════
//  UART RX STATE
// ═══════════════════════════════════════════════════════════════════

uint8_t rxBuf[12];
uint8_t rxPos    = 0;
bool    rxSynced = false;

// ── Flight timers ─────────────────────────────────────────────────
unsigned long waypointStartTime = 0;
bool          waypointTimerActive = false;

unsigned long landingStartTime = 0;
bool          landingActive    = false;

unsigned long rthStartTime = 0;
bool          rthActive    = false;

unsigned long lastTelemetryTime  = 0;
unsigned long lastMetricUpdate   = 0;

// ═══════════════════════════════════════════════════════════════════
//  SIMULATED METRIC UPDATES
// ═══════════════════════════════════════════════════════════════════

void updateGPS() {
  if (simState == SIM_IDLE || simState == SIM_ABORTED) return;
  float dLat = targetLat - simLat;
  float dLng = targetLng - simLng;
  float dist = sqrt(dLat * dLat + dLng * dLng);
  if (dist > 0.000001f) {
    float step = min(dist, dist * 0.12f + GPS_DRIFT_SCALE);
    simLat += (dLat / dist) * step;
    simLng += (dLng / dist) * step;
  }
  simLat += ((float)(random(-10, 10)) / 5000000.0f);
  simLng += ((float)(random(-10, 10)) / 5000000.0f);
}

void updateAltitude() {
  float diff = targetAlt - simAlt;
  if (fabsf(diff) > 0.1f) {
    float rate = (diff > 0) ? 2.0f : -1.5f;
    simAlt += rate * 0.5f;
    simAlt = constrain(simAlt, 0.0f, 200.0f);
  } else {
    simAlt = targetAlt;
  }
}

void updateSpeed() {
  float target = 0.0f;
  if (simState == SIM_FLYING)  target = 5.0f + (float)(random(-5, 15)) / 10.0f;
  if (simState == SIM_RTH)     target = 4.0f;
  if (simState == SIM_LANDING) target = 1.0f;
  simSpeed += (target - simSpeed) * 0.3f;
  simSpeed = max(0.0f, simSpeed);
}

void updateHeading() {
  if (simState == SIM_IDLE || simState == SIM_ABORTED) return;
  float dLat = targetLat - simLat;
  float dLng = targetLng - simLng;
  float bearing = atan2f(dLng, dLat) * 180.0f / PI;
  if (bearing < 0) bearing += 360.0f;
  float diff = bearing - simHeading;
  if (diff > 180)  diff -= 360;
  if (diff < -180) diff += 360;
  simHeading += constrain(diff, -10.0f, 10.0f);
  if (simHeading < 0)    simHeading += 360;
  if (simHeading >= 360) simHeading -= 360;
  simHeading += (float)(random(-5, 5)) / 10.0f;
}

void updateBattery() {
  if ((simState == SIM_FLYING || simState == SIM_RTH) && simBattery > 0) {
    if (random(100) < 5) simBattery--;
  }
}

void updateSignal() {
  simSignal += random(-2, 3);
  simSignal  = constrain(simSignal, -90, -40);
}

void updateAllMetrics() {
  updateGPS();
  updateAltitude();
  updateSpeed();
  updateHeading();
  updateBattery();
  updateSignal();
}

// ═══════════════════════════════════════════════════════════════════
//  UART TX — Responses to Master
// ═══════════════════════════════════════════════════════════════════

void sendNextRequest() {
  Serial2.write(RESP_NEXT_REQ);
  Serial.println(F("[UART TX] 0x06 NEXT_REQUEST"));
}

void sendHomePoint() {
  uint8_t pkt[10];
  pkt[0] = RESP_HOME_POINT;
  memcpy(&pkt[1], &homeLat, 4);
  memcpy(&pkt[5], &homeLng, 4);
  pkt[9] = homeAlt;
  Serial2.write(pkt, 10);
  Serial.printf("[UART TX] 0x07 HOME_POINT (%.6f, %.6f, %dm)\n",
                homeLat, homeLng, homeAlt);
}

void sendArrivedHome() {
  Serial2.write(RESP_ARRIVED_HOME);
  Serial.println(F("[UART TX] 0x08 ARRIVED_HOME"));
}

void sendLanded() {
  Serial2.write(RESP_LANDED);
  Serial.println(F("[UART TX] 0x09 LANDED"));
}

void sendNack() {
  Serial2.write(RESP_NACK);
  Serial.println(F("[UART TX] 0x15 NACK"));
}

// ── 0x0A TELEMETRY — 24-byte packet ──────────────────────────────
// Format: 0x0A + lat(4)+lng(4)+alt(4)+spd(4)+hdg(4)+bat(1)+sig(2) = 24 bytes
void sendTelemetry() {
  uint8_t pkt[24];
  uint8_t* p = pkt;

  *p++ = RESP_TELEMETRY;
  memcpy(p, &simLat,     4); p += 4;
  memcpy(p, &simLng,     4); p += 4;
  memcpy(p, &simAlt,     4); p += 4;
  memcpy(p, &simSpeed,   4); p += 4;
  memcpy(p, &simHeading, 4); p += 4;
  *p++ = simBattery;
  memcpy(p, &simSignal,  2);

  Serial2.write(pkt, 24);

  Serial.printf("[UART TX] 0x0A TELEMETRY lat=%.6f lng=%.6f alt=%.1f spd=%.1f hdg=%.0f bat=%d%% sig=%d\n",
    simLat, simLng, simAlt, simSpeed, simHeading, simBattery, simSignal);
}

// ═══════════════════════════════════════════════════════════════════
//  Checksum verify (XOR bytes 1..10, same as master)
// ═══════════════════════════════════════════════════════════════════

uint8_t calcChecksum(uint8_t* pkt) {
  uint8_t cs = 0;
  for (int i = 1; i <= 10; i++) cs ^= pkt[i];
  return cs;
}

// ═══════════════════════════════════════════════════════════════════
//  Handle complete 12-byte packet from master
// ═══════════════════════════════════════════════════════════════════

void handleMasterPacket(uint8_t* pkt) {
  uint8_t cmd      = pkt[1];
  float   lat, lng;
  uint8_t alt      = pkt[10];
  uint8_t checksum = pkt[11];

  memcpy(&lat, &pkt[2], 4);
  memcpy(&lng, &pkt[6], 4);

  if (calcChecksum(pkt) != checksum) {
    Serial.printf("[UART RX] BAD CHECKSUM expected=0x%02X got=0x%02X — NACK\n",
                  calcChecksum(pkt), checksum);
    sendNack();
    return;
  }

  Serial.printf("[UART RX] CMD=0x%02X lat=%.6f lng=%.6f alt=%d cs=OK\n",
                cmd, lat, lng, alt);

  switch (cmd) {

    case CMD_WAYPOINT:
      targetLat = lat;
      targetLng = lng;
      targetAlt = (float)alt;
      simState  = SIM_FLYING;
      waypointStartTime   = millis();
      waypointTimerActive = true;
      landingActive = false;
      rthActive     = false;
      Serial.printf("[SIM] Flying to: %.6f, %.6f, %dm\n", lat, lng, alt);
      break;

    case CMD_RTH:
      // Master sends RTH as CMD_WAYPOINT with home coords
      // but handle explicit CMD_RTH defensively
      targetLat    = lat;
      targetLng    = lng;
      targetAlt    = (float)alt;
      simState     = SIM_RTH;
      rthStartTime = millis();
      rthActive    = true;
      waypointTimerActive = false;
      landingActive       = false;
      Serial.println(F("[SIM] RTH command"));
      break;

    case CMD_LAND:
      targetAlt        = 0.0f;
      targetLat        = simLat;   // land in place
      targetLng        = simLng;
      simState         = SIM_LANDING;
      landingStartTime = millis();
      landingActive    = true;
      waypointTimerActive = false;
      rthActive           = false;
      Serial.println(F("[SIM] LAND — descending"));
      break;

    case CMD_ABORT:
      simState            = SIM_ABORTED;
      waypointTimerActive = false;
      rthActive           = false;
      landingActive       = false;
      simSpeed            = 0.0f;
      targetAlt           = 0.0f;
      Serial.println(F("[SIM] ABORT"));
      break;

    default:
      Serial.printf("[UART RX] Unknown CMD 0x%02X — NACK\n", cmd);
      sendNack();
      break;
  }
}

// ═══════════════════════════════════════════════════════════════════
//  UART RX Parser
// ═══════════════════════════════════════════════════════════════════

void readMasterUart() {
  while (Serial2.available()) {
    uint8_t b = Serial2.read();
    if (!rxSynced) {
      if (b == UART_START) {
        rxBuf[0] = b;
        rxPos    = 1;
        rxSynced = true;
      }
    } else {
      rxBuf[rxPos++] = b;
      if (rxPos == 12) {
        rxSynced = false;
        rxPos    = 0;
        handleMasterPacket(rxBuf);
      }
    }
  }
}

// ═══════════════════════════════════════════════════════════════════
//  SIMULATION TICK — drives state machine timers
// ═══════════════════════════════════════════════════════════════════

void simulationTick() {
  unsigned long now = millis();

  // Flying → arrived at waypoint
  if (simState == SIM_FLYING && waypointTimerActive) {
    if (now - waypointStartTime >= WAYPOINT_TRAVEL_MS) {
      waypointTimerActive = false;
      simLat = targetLat;
      simLng = targetLng;
      simAlt = targetAlt;
      if (simBattery >= BATTERY_DRAIN_PER_WP)
        simBattery -= BATTERY_DRAIN_PER_WP;
      Serial.println(F("[SIM] Waypoint ARRIVED → NEXT_REQUEST"));
      sendNextRequest();
    }
  }

  // RTH → arrived home
  if (simState == SIM_RTH && rthActive) {
    if (now - rthStartTime >= RTH_TRAVEL_MS) {
      rthActive  = false;
      simLat     = homeLat;
      simLng     = homeLng;
      simAlt     = (float)homeAlt;
      Serial.println(F("[SIM] Arrived home → ARRIVED_HOME"));
      sendArrivedHome();
      // Master will follow with CMD_LAND at home coords
    }
  }

  // Landing → touchdown
  if (simState == SIM_LANDING && landingActive) {
    if (now - landingStartTime >= LANDING_DURATION_MS) {
      landingActive = false;
      simAlt        = 0.0f;
      simSpeed      = 0.0f;
      simState      = SIM_IDLE;
      Serial.println(F("[SIM] Touchdown → LANDED"));
      sendLanded();
    }
  }
}

// ═══════════════════════════════════════════════════════════════════
//  SETUP
// ═══════════════════════════════════════════════════════════════════

void setup() {
  Serial.begin(115200);
  delay(1000);
  randomSeed(analogRead(0));

  Serial.println(F(""));
  Serial.println(F("╔══════════════════════════════════════════╗"));
  Serial.println(F("║   ENISOLAR Slave FC — SIMULATOR MODE    ║"));
  Serial.println(F("╚══════════════════════════════════════════╝"));
  Serial.println(F("[INFO] Telemetry via UART 0x0A (no ESP-NOW)"));

  // UART2 to master
  Serial2.begin(MASTER_BAUD, SERIAL_8N1, MASTER_RX_PIN, MASTER_TX_PIN);
  Serial.println(F("[UART] Serial2 ready (RX=16, TX=17)"));

  // Simulate GPS lock delay
  Serial.println(F("[SIM] Acquiring GPS lock..."));
  delay(3000);

  // Send HOME_POINT to master (first GPS fix)
  homeLat = simLat;
  homeLng = simLng;
  homeAlt = 0;
  sendHomePoint();

  Serial.println(F("[SIM] GPS locked. Home sent. Waiting for waypoints."));
}

// ═══════════════════════════════════════════════════════════════════
//  LOOP
// ═══════════════════════════════════════════════════════════════════

void loop() {
  unsigned long now = millis();

  readMasterUart();
  simulationTick();

  // Update simulated metrics every 500ms
  if (now - lastMetricUpdate >= 500) {
    lastMetricUpdate = now;
    updateAllMetrics();
  }

  // Send telemetry to master via UART every 3s
  if (now - lastTelemetryTime >= TELEMETRY_INTERVAL_MS) {
    lastTelemetryTime = now;
    sendTelemetry();
  }

  delay(10);
}
