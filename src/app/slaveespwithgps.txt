// ═══════════════════════════════════════════════════════════════════
//  ENISOLAR — Slave FC  |  ESP32-WROOM
//
//  ┌─────────────────────────────────────────────────────────────┐
//  │  WIRING                                                     │
//  │                                                             │
//  │  ── NEO-6M GPS  (UART1 / gpsSerial) ───────────────────── │
//  │     NEO-6M TX  →  ESP32 GPIO 26  (RX)                      │
//  │     NEO-6M RX  →  ESP32 GPIO 27  (TX, optional)            │
//  │     NEO-6M VCC →  ESP32 3V3                                │
//  │     NEO-6M GND →  ESP32 GND                                │
//  │                                                             │
//  │  ── MASTER ↔ SLAVE  (UART2 / Serial2, crossed) ─────────── │
//  │     Slave  GPIO 16 (RX2)  ←  Master GPIO 17 (TX2)          │
//  │     Slave  GPIO 17 (TX2)  →  Master GPIO 16 (RX2)          │
//  │     GND ─────────────────────── GND                        │
//  └─────────────────────────────────────────────────────────────┘
//
//  UART MAP:
//    Serial  (UART0) — USB debug @ 115200  (GPIO3/GPIO1)
//    UART1   (gpsSerial) — NEO-6M GPS @ 9600  (GPIO26 RX, GPIO27 TX)
//    Serial2 (UART2) — Master comms @ 115200  (GPIO16 RX, GPIO17 TX)
//
// ═══════════════════════════════════════════════════════════════════

#include <math.h>
#include <HardwareSerial.h>

// ── UART1 — NEO-6M GPS ───────────────────────────────────────────
#define GPS_RX_PIN  26
#define GPS_TX_PIN  27
#define GPS_BAUD    9600
HardwareSerial gpsSerial(1);   // UART1 remapped to GPIO26/27

// ── UART2 — Master comms ─────────────────────────────────────────
#define MASTER_RX_PIN  16
#define MASTER_TX_PIN  17
#define MASTER_BAUD    115200

// ── Packet bytes (Master → Slave) ────────────────────────────────
#define UART_START    0xAA
#define CMD_WAYPOINT  0x01
#define CMD_RTH       0x02
#define CMD_LAND      0x03
#define CMD_ABORT     0x04

// ── Response bytes (Slave → Master) ──────────────────────────────
//   HOME_POINT : 0x07 + lat(4) + lng(4) + alt(1)         = 10 bytes
//   TELEMETRY  : 0x0A + lat(4)+lng(4)+alt(4)+spd(4)+hdg(4)
//                     + bat(1) + sig(2)                    = 24 bytes
#define RESP_NEXT_REQ      0x06
#define RESP_HOME_POINT    0x07
#define RESP_ARRIVED_HOME  0x08
#define RESP_LANDED        0x09
#define RESP_TELEMETRY     0x0A
#define RESP_NACK          0x15
#define TELEMETRY_PAYLOAD_LEN  24

// ── Timing ────────────────────────────────────────────────────────
#define WAYPOINT_TRAVEL_MS     6000
#define LANDING_DURATION_MS    4000
#define RTH_TRAVEL_MS          8000
#define TELEMETRY_INTERVAL_MS  3000
#define BATTERY_DRAIN_PER_WP   1

// ═══════════════════════════════════════════════════════════════════
//  STATE MACHINE
// ═══════════════════════════════════════════════════════════════════

enum SlaveState { SIM_IDLE, SIM_FLYING, SIM_LANDING, SIM_RTH, SIM_ABORTED };
SlaveState simState = SIM_IDLE;

// ═══════════════════════════════════════════════════════════════════
//  LIVE GPS — NEO-6M NMEA on UART1 (GPIO26/27)
//  Parser taken from confirmed-working test code:
//  - strstr("GGA") to match any GGA variant ($GPGGA, $GNGGA, etc.)
//  - Hand-built comma splitter that handles empty fields (,,)
//  - field[0]=$GPGGA [1]=time [2]=lat [3]=N/S [4]=lng [5]=E/W
//    [6]=quality [7]=sats [8]=HDOP [9]=alt [10]=M
// ═══════════════════════════════════════════════════════════════════

float   gpsLat   = 0.0f;
float   gpsLng   = 0.0f;
float   gpsAlt   = 0.0f;
uint8_t gpsSats  = 0;
bool    gpsFixed = false;

char    nmeaBuf[128];
uint8_t nmeaPos     = 0;
bool    nmeaInFrame = false;

float nmeaToDecimal(float raw, char hemi) {
  int   deg  = (int)(raw / 100.0f);
  float mins = raw - (float)(deg * 100);
  float dd   = (float)deg + mins / 60.0f;
  if (hemi == 'S' || hemi == 'W') dd = -dd;
  return dd;
}

void parseNMEA(const char* sentence) {
  if (strstr(sentence, "GGA") == nullptr) return;

  char buf[128];
  strncpy(buf, sentence, sizeof(buf) - 1);
  buf[sizeof(buf) - 1] = '\0';

  // Hand-built splitter — handles empty fields (,,) unlike strtok
  char*  fields[16];
  int    fc = 0;
  char*  p  = buf;
  while (fc < 16) {
    fields[fc++] = p;
    p = strchr(p, ',');
    if (!p) break;
    *p++ = '\0';
  }

  if (fc < 10) return;

  // fields[6] = fix quality (0 = no fix)
  uint8_t quality = (uint8_t)atoi(fields[6]);
  if (quality > 0) {
    gpsLat  = nmeaToDecimal(atof(fields[2]), fields[3][0]);
    gpsLng  = nmeaToDecimal(atof(fields[4]), fields[5][0]);
    gpsSats = (uint8_t)atoi(fields[7]);
    gpsAlt  = atof(fields[9]);
    gpsFixed = true;
    Serial.printf("[GPS] Fix: %.6f, %.6f, %.1fm  sats=%d\n",
                  gpsLat, gpsLng, gpsAlt, gpsSats);
  } else {
    gpsFixed = false;
    Serial.println(F("[GPS] No fix"));
  }
}

void readGPS() {
  while (gpsSerial.available()) {
    char c = (char)gpsSerial.read();
    if (c == '$') { nmeaInFrame = true; nmeaPos = 0; }
    if (nmeaInFrame) {
      if (c == '\n' || c == '\r') {
        nmeaBuf[nmeaPos] = '\0';
        if (nmeaPos > 5) parseNMEA(nmeaBuf);
        nmeaInFrame = false;
        nmeaPos     = 0;
      } else {
        if (nmeaPos < sizeof(nmeaBuf) - 1)
          nmeaBuf[nmeaPos++] = c;
      }
    }
  }
}

// ═══════════════════════════════════════════════════════════════════
//  DERIVED SENSOR VALUES
//  Speed & heading derived from GPS positions.
//  Battery & signal simulated until real hardware wired.
// ═══════════════════════════════════════════════════════════════════

float   simSpeed   = 0.0f;
float   simHeading = 0.0f;
uint8_t simBattery = 100;
int16_t simSignal  = -55;

float         prevLat        = 0.0f;
float         prevLng        = 0.0f;
unsigned long prevMetricTime = 0;

void updateSpeed() {
  if (!gpsFixed) { simSpeed = 0.0f; return; }
  unsigned long now = millis();
  float dt = (now - prevMetricTime) / 1000.0f;
  if (dt < 0.1f) return;
  float dLat = (gpsLat - prevLat) * 111320.0f;
  float dLng = (gpsLng - prevLng) * 111320.0f * cosf(gpsLat * DEG_TO_RAD);
  simSpeed   = sqrtf(dLat * dLat + dLng * dLng) / dt;
  simSpeed   = constrain(simSpeed, 0.0f, 30.0f);
  prevLat = gpsLat; prevLng = gpsLng; prevMetricTime = now;
}

void updateHeading(float tgtLat, float tgtLng) {
  if (!gpsFixed) return;
  float bearing = atan2f(tgtLng - gpsLng, tgtLat - gpsLat) * RAD_TO_DEG;
  if (bearing < 0) bearing += 360.0f;
  float diff = bearing - simHeading;
  if (diff >  180) diff -= 360;
  if (diff < -180) diff += 360;
  simHeading += constrain(diff, -10.0f, 10.0f);
  if (simHeading <   0) simHeading += 360;
  if (simHeading >= 360) simHeading -= 360;
}

void updateBattery() {
  if ((simState == SIM_FLYING || simState == SIM_RTH) && simBattery > 0)
    if (random(100) < 5) simBattery--;
}

void updateSignal() {
  simSignal += random(-2, 3);
  simSignal  = constrain(simSignal, -90, -40);
}

// ═══════════════════════════════════════════════════════════════════
//  TARGET / HOME
// ═══════════════════════════════════════════════════════════════════

float   targetLat = 0.0f;
float   targetLng = 0.0f;
float   targetAlt = 0.0f;

float   homeLat   = 0.0f;
float   homeLng   = 0.0f;
uint8_t homeAlt   = 0;
bool    homeSet   = false;

// ═══════════════════════════════════════════════════════════════════
//  UART2 RX STATE — 12-byte packets from master
// ═══════════════════════════════════════════════════════════════════

uint8_t rxBuf[12];
uint8_t rxPos    = 0;
bool    rxSynced = false;

unsigned long waypointStartTime   = 0;
bool          waypointTimerActive = false;
unsigned long landingStartTime    = 0;
bool          landingActive       = false;
unsigned long rthStartTime        = 0;
bool          rthActive           = false;

unsigned long lastTelemetryTime = 0;
unsigned long lastMetricUpdate  = 0;

// ═══════════════════════════════════════════════════════════════════
//  TX — Responses to Master over Serial2
// ═══════════════════════════════════════════════════════════════════

void sendNextRequest() {
  Serial2.write(RESP_NEXT_REQ);
  Serial.println(F("[TX] 0x06 NEXT_REQUEST"));
}

void sendHomePoint() {
  uint8_t pkt[10];
  pkt[0] = RESP_HOME_POINT;
  memcpy(&pkt[1], &homeLat, 4);
  memcpy(&pkt[5], &homeLng, 4);
  pkt[9] = homeAlt;
  Serial2.write(pkt, 10);
  Serial.printf("[TX] 0x07 HOME_POINT (%.6f, %.6f, %dm)\n",
                homeLat, homeLng, homeAlt);
}

void sendArrivedHome() {
  Serial2.write(RESP_ARRIVED_HOME);
  Serial.println(F("[TX] 0x08 ARRIVED_HOME"));
}

void sendLanded() {
  Serial2.write(RESP_LANDED);
  Serial.println(F("[TX] 0x09 LANDED"));
}

void sendNack() {
  Serial2.write(RESP_NACK);
  Serial.println(F("[TX] 0x15 NACK"));
}

void sendTelemetry() {
  if (!gpsFixed) { Serial.println(F("[TELEM] No fix — skip")); return; }

  uint8_t pkt[TELEMETRY_PAYLOAD_LEN];
  uint8_t* p = pkt;
  *p++ = RESP_TELEMETRY;
  memcpy(p, &gpsLat,     4); p += 4;
  memcpy(p, &gpsLng,     4); p += 4;
  memcpy(p, &gpsAlt,     4); p += 4;
  memcpy(p, &simSpeed,   4); p += 4;
  memcpy(p, &simHeading, 4); p += 4;
  *p++ = simBattery;
  memcpy(p, &simSignal,  2);

  Serial2.write(pkt, TELEMETRY_PAYLOAD_LEN);
  Serial.printf("[TX] 0x0A TELEM lat=%.6f lng=%.6f alt=%.1f "
                "spd=%.1f hdg=%.0f bat=%d sig=%d\n",
                gpsLat, gpsLng, gpsAlt,
                simSpeed, simHeading, simBattery, simSignal);
}

// ═══════════════════════════════════════════════════════════════════
//  CHECKSUM
//  Mirrors master's calcChecksum(cmd, lat, lng, alt):
//  cs = cmd ^ lat_bytes[0..3] ^ lng_bytes[0..3] ^ alt
//  Packet: [0]=0xAA [1]=cmd [2..5]=lat [6..9]=lng [10]=alt [11]=cs
// ═══════════════════════════════════════════════════════════════════

uint8_t calcChecksum(uint8_t* pkt) {
  uint8_t cs = pkt[1];
  for (int i = 2; i <= 5;  i++) cs ^= pkt[i];
  for (int i = 6; i <= 9;  i++) cs ^= pkt[i];
  cs ^= pkt[10];
  return cs;
}

// ═══════════════════════════════════════════════════════════════════
//  Handle complete 12-byte packet from master
// ═══════════════════════════════════════════════════════════════════

void handleMasterPacket(uint8_t* pkt) {
  uint8_t cmd = pkt[1], alt = pkt[10], checksum = pkt[11];
  float lat, lng;
  memcpy(&lat, &pkt[2], 4);
  memcpy(&lng, &pkt[6], 4);

  uint8_t expected = calcChecksum(pkt);
  if (expected != checksum) {
    Serial.printf("[RX] BAD CS exp=0x%02X got=0x%02X — NACK\n",
                  expected, checksum);
    sendNack();
    return;
  }

  Serial.printf("[RX] CMD=0x%02X lat=%.6f lng=%.6f alt=%d OK\n",
                cmd, lat, lng, alt);

  switch (cmd) {
    case CMD_WAYPOINT:
      targetLat = lat; targetLng = lng; targetAlt = (float)alt;
      simState  = SIM_FLYING;
      waypointStartTime = millis(); waypointTimerActive = true;
      landingActive = false; rthActive = false;
      Serial.printf("[FC] Fly to: %.6f, %.6f, %dm\n", lat, lng, alt);
      break;

    case CMD_RTH:
      targetLat = lat; targetLng = lng; targetAlt = (float)alt;
      simState  = SIM_RTH;
      rthStartTime = millis(); rthActive = true;
      waypointTimerActive = false; landingActive = false;
      Serial.println(F("[FC] RTH"));
      break;

    case CMD_LAND:
      targetAlt = 0.0f;
      targetLat = gpsFixed ? gpsLat : lat;
      targetLng = gpsFixed ? gpsLng : lng;
      simState  = SIM_LANDING;
      landingStartTime = millis(); landingActive = true;
      waypointTimerActive = false; rthActive = false;
      Serial.println(F("[FC] LAND"));
      break;

    case CMD_ABORT:
      simState = SIM_ABORTED;
      waypointTimerActive = false; rthActive = false; landingActive = false;
      simSpeed = 0.0f; targetAlt = 0.0f;
      Serial.println(F("[FC] ABORT"));
      break;

    default:
      Serial.printf("[RX] Unknown CMD 0x%02X — NACK\n", cmd);
      sendNack();
      break;
  }
}

// ═══════════════════════════════════════════════════════════════════
//  UART2 RX Parser
// ═══════════════════════════════════════════════════════════════════

void readMasterUart() {
  while (Serial2.available()) {
    uint8_t b = Serial2.read();
    if (!rxSynced) {
      if (b == UART_START) { rxBuf[0] = b; rxPos = 1; rxSynced = true; }
    } else {
      rxBuf[rxPos++] = b;
      if (rxPos == 12) {
        rxSynced = false; rxPos = 0;
        handleMasterPacket(rxBuf);
      }
    }
  }
}

// ═══════════════════════════════════════════════════════════════════
//  FLIGHT STATE MACHINE TICK
// ═══════════════════════════════════════════════════════════════════

void simulationTick() {
  unsigned long now = millis();

  if (simState == SIM_FLYING && waypointTimerActive &&
      now - waypointStartTime >= WAYPOINT_TRAVEL_MS) {
    waypointTimerActive = false;
    if (simBattery >= BATTERY_DRAIN_PER_WP) simBattery -= BATTERY_DRAIN_PER_WP;
    Serial.println(F("[FC] Arrived → NEXT_REQUEST"));
    sendNextRequest();
  }

  if (simState == SIM_RTH && rthActive &&
      now - rthStartTime >= RTH_TRAVEL_MS) {
    rthActive = false;
    Serial.println(F("[FC] At home → ARRIVED_HOME"));
    sendArrivedHome();
  }

  if (simState == SIM_LANDING && landingActive &&
      now - landingStartTime >= LANDING_DURATION_MS) {
    landingActive = false;
    simSpeed = 0.0f;
    simState = SIM_IDLE;
    Serial.println(F("[FC] Touchdown → LANDED"));
    sendLanded();
  }
}

// ═══════════════════════════════════════════════════════════════════
//  SETUP
// ═══════════════════════════════════════════════════════════════════

void setup() {
  // UART0 — USB debug (GPIO3/GPIO1, USB available since GPS moved to GPIO26/27)
  Serial.begin(115200);
  delay(500);

  // UART1 — NEO-6M GPS (GPIO26 RX, GPIO27 TX)
  gpsSerial.begin(GPS_BAUD, SERIAL_8N1, GPS_RX_PIN, GPS_TX_PIN);

  // UART2 — Master comms (GPIO16 RX, GPIO17 TX)
  Serial2.begin(MASTER_BAUD, SERIAL_8N1, MASTER_RX_PIN, MASTER_TX_PIN);

  randomSeed(analogRead(34));

  Serial.println(F(""));
  Serial.println(F("╔════════════════════════════════════════════════╗"));
  Serial.println(F("║  ENISOLAR Slave FC — NEO-6M GPS (GPIO26/27)   ║"));
  Serial.println(F("╚════════════════════════════════════════════════╝"));
  Serial.println(F("[INFO] GPS=UART1(GPIO26/27)  Master=UART2(GPIO16/17)"));

  Serial.println(F("[GPS] Waiting for first fix (30-90s cold start)..."));
  unsigned long gpsWait = millis();
  while (!gpsFixed) {
    readGPS();
    if (millis() - gpsWait >= 120000UL) {
      Serial.println(F("[GPS] WARNING: No fix after 120s — continuing"));
      break;
    }
    delay(10);
  }

  if (gpsFixed) {
    homeLat = gpsLat;
    homeLng = gpsLng;
    homeAlt = (uint8_t)constrain((int)gpsAlt, 0, 255);
    homeSet = true;
    prevLat = gpsLat; prevLng = gpsLng;
    prevMetricTime = millis();
    sendHomePoint();
    Serial.printf("[FC] Home locked: %.6f, %.6f, %dm\n",
                  homeLat, homeLng, homeAlt);
  } else {
    Serial.println(F("[FC] No GPS fix — home NOT sent. Master will block RTH."));
  }
}

// ═══════════════════════════════════════════════════════════════════
//  LOOP
// ═══════════════════════════════════════════════════════════════════

void loop() {
  unsigned long now = millis();

  // 1. Read NMEA from NEO-6M on UART1 (GPIO26)
  readGPS();

  // 2. First fix after boot — lock home, notify master
  if (gpsFixed && !homeSet) {
    homeLat = gpsLat; homeLng = gpsLng;
    homeAlt = (uint8_t)constrain((int)gpsAlt, 0, 255);
    homeSet = true;
    prevLat = gpsLat; prevLng = gpsLng;
    prevMetricTime = now;
    sendHomePoint();
    Serial.printf("[FC] Late fix — home locked: %.6f, %.6f, %dm\n",
                  homeLat, homeLng, homeAlt);
  }

  // 3. Process master commands from UART2
  readMasterUart();

  // 4. Flight state machine
  simulationTick();

  // 5. Derived metrics every 500 ms
  if (now - lastMetricUpdate >= 500) {
    lastMetricUpdate = now;
    updateSpeed();
    updateHeading(targetLat, targetLng);
    updateBattery();
    updateSignal();
  }

  // 6. Telemetry to master every 3 s (only when GPS fixed)
  if (gpsFixed && (now - lastTelemetryTime >= TELEMETRY_INTERVAL_MS)) {
    lastTelemetryTime = now;
    sendTelemetry();
  }

  delay(10);
}
